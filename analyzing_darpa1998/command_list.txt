#######################################################################
# EXERCISE 4: TURN PACKETS INTO KDD STYLE RECORDS
#######################################################################

# Slide 68 · Step 32 of 44 · Point A and B back at neptune
export A=230.1.10.20
export B=172.16.112.50

# Slide 69 · Step 33 of 44 · Save the flood into its own file
# Takes a while: reads the whole 430 MB capture.
tshark -r friday.tcpdump -Y "ip.src == $A && ip.dst == $B" -w neptune.pcap

# Slide 70 · Step 34 of 44 · Take the first 3,000 packets
tshark -r neptune.pcap -c 3000 -w neptune_small.pcap

# Slide 71 · Step 35 of 44 · Confirm the slice size
tshark -r neptune_small.pcap -T fields -e frame.number | wc -l

# Slide 72 · Step 36 of 44 · Flatten the slice into a table
# One command split over five lines. Copy all five lines together.
tshark -r neptune_small.pcap -T fields -E separator=, -E occurrence=f \
  -e frame.time_epoch -e ip.src -e ip.dst -e ip.proto \
  -e tcp.srcport -e tcp.dstport -e udp.srcport -e udp.dstport \
  -e tcp.flags -e tcp.len -e udp.length -e icmp.type \
  > packets.csv

# Slide 73 · Step 37 of 44 · Look at the table
head packets.csv

# Slide 75 · Step 38 of 44 · Build the connection records
python3 mini_kdd.py packets.csv > records.csv

# Slide 76 · Step 39 of 44 · Read the records you built
head records.csv

# Slide 78 · Step 40 of 44 · Count the records by flag
cut -d, -f5 records.csv | sort | uniq -c | sort -rn
